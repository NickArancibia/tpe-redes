\section{El problema}

\begin{frame}{The Store es nuestra tienda online}
  \centering
  \vfill
  {\Large Si la tienda no abre, \alert{no vendemos}.}
  \vfill
  The Store es el canal de venta online de nuestra empresa.\\
  Por eso tiene que estar siempre disponible.
  \vfill
\end{frame}
\note{
  \guion{\oradorA}{0:45}
  \begin{itemize}
    \item Hablar en primera persona, como el equipo que opera la tienda.
    \item No explicar cómo está hecha la tienda: lo único que importa acá es que, si no abre,
      se pierden ventas.
  \end{itemize}
}

\begin{frame}{Llega el Black Friday}
  \begin{description}[00:00]
    \item[00:00] Arranca la promo. Entran muchos más clientes que en un día normal.
    \item[00:10] La tienda se pone lenta: hace falta sumar capacidad.
    \item[00:15] Para sumarla hay que \textbf{cerrar la tienda} y volver a armarla desde cero.
    \item[00:20] Los pasos no están escritos en ningún lado: depende de quién esté de guardia.
  \end{description}
  \bigskip
  \alert{La tienda cierra justo cuando más clientes hay.}
\end{frame}
\note{
  \guion{\oradorA}{1:15}
  \begin{itemize}
    \item Contarlo como una historia, en presente: ``son las doce de la noche y arranca el Black
      Friday\ldots''.
    \item El problema no es la tienda sino cómo la armamos y la operamos: hoy, crecer significa cerrar.
  \end{itemize}
}

\begin{frame}{No es solo el Black Friday}
  \begin{columns}[T]
    \column{0.31\textwidth}
      \textbf{Mantenimiento}\par\smallskip
      \small Un servidor necesita mantenimiento. Para sacarlo, hay que cerrar la tienda.
    \column{0.31\textwidth}
      \textbf{Una falla}\par\smallskip
      \small Algo se rompe y hay que rearmar todo de memoria, sin saber cuánto va a tardar.
    \column{0.31\textwidth}
      \textbf{Picos de tráfico}\par\smallskip
      \small Baja el tráfico, pero sacar servidores también obliga a cerrar. Seguimos pagando
      lo que no usamos.
  \end{columns}
  \vfill
  \centering
  \alert{Hoy, cada cambio en la tienda significa cerrarla.}
  \vfill
\end{frame}
\note{
  \guion{\oradorA}{1:00}
  \begin{itemize}
    \item Tres situaciones de todos los días, no solo de Black Friday.
    \item Remarcar la frase final: es el problema que venimos a resolver.
  \end{itemize}
}

\begin{frame}{Caso 1 · Armar todo desde cero}
  \caso{Una computadora sin nada de la tienda.}
       {Ejecutamos la receta que arma los servidores. Después, la ejecutamos de nuevo.}
       {Aparecen tres servidores listos. La segunda vez no cambia nada: la receta solo hace lo
        que falta.}
\end{frame}
\note{
  \guion{\oradorB}{2:30}
  \begin{itemize}
    \item Comandos: \cmd{ansible-playbook create-cluster.yml} (dos veces), \cmd{docker ps},
      \cmd{kubectl get nodes -o wide}.
    \item Mientras corre: ``estamos armando tres servidores''. Señalar cuando pasan a Ready:
      ``listos''.
    \item Segunda ejecución: zoom en el resumen final, \texttt{changed=0}: ``no tuvo nada que
      hacer, ya estaba todo''.
  \end{itemize}
}

\begin{frame}{Caso 2 · Abrir la tienda}
  \caso{Los tres servidores del caso 1, todavía sin la tienda.}
       {Ejecutamos la receta que instala la tienda.}
       {La tienda abre en el navegador, entrando por cualquiera de los servidores, y su trabajo
        queda repartido entre ellos.}
\end{frame}
\note{
  \guion{\oradorB}{2:30}
  \begin{itemize}
    \item Comandos: \cmd{ansible-playbook deploy-store.yml},
      \cmd{kubectl get pods -n the-store -o wide}.
    \item Abrir la tienda por dos direcciones distintas (172.28.0.11 y 172.28.0.12): ``entra por
      cualquier servidor''.
    \item Columna NODE: las partes de la tienda quedaron repartidas entre los servidores.
  \end{itemize}
}

\begin{frame}{Caso 3 · Llega el Black Friday: sumar un servidor}
  \caso{La tienda del caso 2, abierta.}
       {Agregamos un servidor a la lista y ejecutamos la receta para sumar servidores.}
       {Entra el servidor nuevo. Los que ya estaban no se tocan y la tienda nunca cierra.}
\end{frame}
\note{
  \guion{\oradorB}{1:45}
  \begin{itemize}
    \item Agregar \texttt{k3s-agent-3} (172.28.0.13) al inventario en vivo y mostrar
      \cmd{git diff}: ``crecer es agregar un renglón a la lista''.
    \item Comandos: \cmd{ansible-playbook add-nodes.yml}, \cmd{kubectl get nodes}.
    \item Resumen final: solo el nodo nuevo tiene cambios; los existentes, en \texttt{ok}.
    \item Si preguntan: el nodo nuevo no recibe trabajo de inmediato (el scheduler prefiere nodos
      que ya tienen las imágenes). Lo vamos a ver recibir trabajo en el caso 4.
    \item Pase a \oradorC.
  \end{itemize}
}

\begin{frame}{Caso 4 · Retirar un servidor}
  \caso{Los cuatro servidores del caso 3, con la tienda abierta.}
       {Ejecutamos la receta para retirar un servidor y lo sacamos de la lista.}
       {Su trabajo pasa a los otros servidores, el servidor desaparece y la tienda sigue
        abierta.}
\end{frame}
\note{
  \guion{\oradorC}{2:15}
  \begin{itemize}
    \item Comandos: \cmd{ansible-playbook remove-node.yml -e node=k3s-agent-1}, después sacar
      \texttt{k3s-agent-1} del inventario; \cmd{kubectl get nodes},
      \cmd{kubectl get pods -n the-store -o wide}, \cmd{docker ps}.
    \item Narrar: ``le avisamos que no reciba más trabajo, pasamos su trabajo a los otros y lo
      apagamos'' (cordon, drain, borrar el nodo, eliminar contenedor y volumen).
    \item Avisar antes: hay una sola copia de cada parte, así que la tienda puede tardar unos
      segundos en responder.
  \end{itemize}
}

\begin{frame}{Caso 5 · Se rompe todo: reconstruir}
  \caso{La tienda funcionando.}
       {Borramos todo. Después ejecutamos las recetas para armar los servidores y abrir la
        tienda.}
       {Después de borrar no queda nada. Después de reconstruir, todo queda como al final de
        los casos 1 y 2.}
\end{frame}
\note{
  \guion{\oradorC}{2:30}
  \begin{itemize}
    \item Comandos: \cmd{ansible-playbook destroy.yml};
      \cmd{docker ps -a; docker network ls; docker volume ls};
      \cmd{ansible-playbook create-cluster.yml}, \cmd{ansible-playbook deploy-store.yml}.
    \item El destroy busca los recursos por la etiqueta del proyecto: borra también nodos que ya
      no están en el inventario.
    \item Mostrar los tres listados vacíos: ``no queda nada''.
    \item Lanzar la reconstrucción y volver a las diapositivas mientras corre (dura TODO minutos).
    \item Aclarar que se reconstruye con la lista actual (server, agent-2 y agent-3).
  \end{itemize}

\begin{frame}{Próximos pasos}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{Hoy, en esta prueba}
      \begin{itemize}\small
        \item Cada parte de la tienda tiene una sola copia: al retirar su servidor, puede tardar
          unos segundos en responder.
        \item Un solo servidor coordina a los demás: si se cae, no se pueden hacer cambios hasta
          que vuelva.
        \item Todo corre en una sola computadora, y la tienda se abre desde ella.
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{Siguiente paso}
      \begin{itemize}\small
        \item Dos o más copias de cada parte.
        \item Tres servidores coordinadores.
        \item Llevarlo a servidores reales o a la nube, cambiando solo la parte que crea los
          servidores, con una única entrada para los clientes.
      \end{itemize}
  \end{columns}
\end{frame}
